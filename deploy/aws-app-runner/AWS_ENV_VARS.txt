# - Do NOT commit real secret values to git. Use placeholders
#   like <REPLACE-ME> in this file and fill in only during deploy.

# === REQUIRED — Core Configuration ===

NODE_ENV=production
PORT=3001                       # App Runner expects this; Dockerfile EXPOSEs 3001
BASE_URL=https://marketplace.teneo.io


# === REQUIRED — Secrets (store in Secrets Manager) ===

# Generate locally: openssl rand -hex 32
# Save as: aws secretsmanager create-secret --name openbazaar-ai/session-secret
SESSION_SECRET=<from-secrets-manager:openbazaar-ai/session-secret>

# Generate locally: node marketplace/backend/scripts/generate-password-hash.js --generate
# Save as: aws secretsmanager create-secret --name openbazaar-ai/admin-password-hash
ADMIN_PASSWORD_HASH=<from-secrets-manager:openbazaar-ai/admin-password-hash>


# === REQUIRED — Database (Supabase Postgres) ===

# Supabase Dashboard → Settings → Database → Connection Pooling URI
# Note: pooler URL on port 6543, NOT direct connection on 5432
DATABASE_URL=<from-secrets-manager:openbazaar-ai/database-url>

# Optional alias (some code paths check this)
SUPABASE_DB_URL=<same-as-DATABASE_URL>


# === REQUIRED — Stripe (restricted key from Teneo's account) ===

# Stripe Dashboard → Developers → API keys → Create restricted key
# Permissions: charges:write, customers:write, payment_intents:write,
#              checkout_sessions:write, webhook_endpoints:read
# Save as: aws secretsmanager create-secret --name openbazaar-ai/stripe-secret-key
STRIPE_SECRET_KEY=<from-secrets-manager:openbazaar-ai/stripe-secret-key>

# Publishable key — public, no restrictions needed
# Same Stripe account as teneo.io (pk_live_*)
STRIPE_PUBLISHABLE_KEY=pk_live_<TENEO_STRIPE_PUBLISHABLE_KEY>

# Webhook signing secret — created in step 3.3 of README-DEPLOY.md
# Stripe Dashboard → Developers → Webhooks → marketplace.teneo.io endpoint → Signing secret
STRIPE_WEBHOOK_SECRET=<from-secrets-manager:openbazaar-ai/stripe-webhook-secret>


# === REQUIRED — Service-key chain (mutual auth with teneo.io) ===

# Pre-shared secret between marketplace.teneo.io ↔ teneo.io Lambdas
# Generate ONCE: openssl rand -hex 32
# Same value lives in:
#   - This file as TENEO_SERVICE_KEYS (used to validate INCOMING calls from teneo.io)
#   - teneo-production marketplace-api Lambda as OPENBAZAAR_SERVICE_KEY (used in OUTGOING calls to here)
TENEO_SERVICE_KEYS=<from-secrets-manager:teneo-marketplace/service-key>

# The same secret, used as the OUTGOING service key when marketplace.teneo.io
# calls BACK to teneo.io (order-paid, credit-author, validate-license callbacks)
TENEO_FULFILLMENT_SERVICE_KEY=<from-secrets-manager:teneo-marketplace/service-key>

# Where marketplace.teneo.io sends callbacks
TENEO_FULFILLMENT_BASE_URL=https://api.teneo.io


# === OPTIONAL — AI features ===

# Without this, AI Discovery falls back to keyword search (still works)
# Use Teneo's existing OpenAI key — same account
OPENAI_API_KEY=<from-secrets-manager:openbazaar-ai/openai-api-key>


# === OPTIONAL — Email (transactional) ===

# Used for: download-link delivery, order receipts, admin notifications
# Recommended: SES via Teneo's existing AWS SES setup
EMAIL_ENABLED=true
EMAIL_PROVIDER=ses                # ses | smtp
EMAIL_FROM=marketplace@teneo.io   # must be verified in SES
EMAIL_REPLY_TO=support@teneo.io

# If using SMTP fallback:
# EMAIL_HOST=smtp.gmail.com
# EMAIL_PORT=587
# EMAIL_USER=<from-secrets-manager:openbazaar-ai/smtp-user>
# EMAIL_PASS=<from-secrets-manager:openbazaar-ai/smtp-app-password>


# === OPTIONAL — Branding ===

MARKETPLACE_NAME=Teneo Marketplace
MARKETPLACE_TAGLINE=Knowledge Beyond Boundaries
SUPPORT_EMAIL=support@teneo.io
SITE_URL=https://marketplace.teneo.io       # back-compat with code expecting this name


# === OPTIONAL — Federation (V3+ work, set to inert defaults now) ===

FEDERATION_PEERS=                          # comma-separated peer URLs, empty for V0
